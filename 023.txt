#include <stdio.h>
int main(void)
{
	int a[3][3];
	int n =1;
	for (int i = 0; i < 3; i++)
		for(int j = 0; j<3; j++)
		a[i][j] = n++;
	for (int i = 0; i < 3; i++) {
		int sum = 0;
		for(int j = 0; j<3; j++) {
		printf("%d", a[i][j]);
		sum += a[i][j];
	}
	printf("%d\n", sum);
}
	
return 0;
}
